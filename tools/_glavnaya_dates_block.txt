     // Функция для форматирования даты в строку DD Month
function formatDate(date) {
    var dd = String(date.getDate()).padStart(2, '0');
    var mm = date.getMonth(); 
    var months = ['янв', 'фев', 'мар', 'апр', 'май', 'июн', 'июл', 'авг', 'сен', 'окт', 'ноя', 'дек'];
    var monthName = months[mm];

    return dd + ' ' + monthName;
}
		
function formatDateAjax(date) {
    var day = date.getDate(); // день
    var month = date.getMonth() + 1; // месяцы считаются с 0 в JavaScript
    var year = date.getFullYear(); // год

    // Если день или месяц меньше 10, добавляем в начало "0"
    if (day < 10) day = '0' + day;
    if (month < 10) month = '0' + month;

    return day + '.' + month + '.' + year;
}

function removeLoader() {

function updateSelectOptions(data) {
    var selectIndex = 1;
    for (var sectionName in data) {
        var selectElement = document.getElementById('select' + selectIndex);
        if (selectElement) {
            selectElement.innerHTML = '';  // очистка текущего содержимого
            data[sectionName].forEach(function(item) {
                var option = document.createElement('option');
                option.value = item.value;
                option.textContent = item.text;
                selectElement.appendChild(option);
            });
        }
        selectIndex++;
    }
}
